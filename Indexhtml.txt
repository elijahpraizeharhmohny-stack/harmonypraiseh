<script>
// ===== CONFIG =====
const REDIRECT_URL = "https://otieu.com/4/8866944";
const MIN_DELAY = 18; // seconds
const MAX_DELAY = 30; // seconds
const CAP_HOURS = 24;
// ==================

const capKey = "redirected_once_24h";
const now = Date.now();
const lastRedirect = localStorage.getItem(capKey);

// Random human-like delay
const delay = Math.floor(Math.random() * (MAX_DELAY - MIN_DELAY + 1)) + MIN_DELAY;

if (lastRedirect && (now - lastRedirect) < CAP_HOURS * 60 * 60 * 1000) {
    document.getElementById("content").innerHTML = `
        <h1>Access Limit Reached</h1>
        <p>You’ve already opened this link within the last 24 hours.</p>
    `;
} else {
    let timeLeft = delay;
    const timerEl = document.getElementById("timer");
    const btn = document.getElementById("manualBtn");

    timerEl.textContent = timeLeft + "s";

    const countdown = setInterval(() => {
        timeLeft--;
        timerEl.textContent = timeLeft + "s";

        // Enable button halfway (human-like)
        if (timeLeft <= Math.ceil(delay / 2)) {
            btn.classList.add("active");
        }

        if (timeLeft <= 0) {
            clearInterval(countdown);
            localStorage.setItem(capKey, now);
            window.location.href = REDIRECT_URL;
        }
    }, 1000);
}
</script>
